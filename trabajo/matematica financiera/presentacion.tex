    \newline\newline

  \begin{frame}
    Partimos de la fórmula del valor futuro y despejamos $P$:

    \begin{equation*}
      \begin{aligned}
        F &= P(1+i)^n \\
        P &= \frac{F}{(1+i)^n} \\
        P &= F(1+i)^{-n}
      \end{aligned}
    \end{equation*}
  \end{frame}

  \begin{frame}
    \textit{Fórmula}: El valor presente de una cantidad $F$ que se recibe dentro
    de $n$ periodos es
    \[P = F(1+i)^{-n}\]

    Llevar un valor al presente se llama descontar.
  \end{frame}

  \begin{frame}
    \textit{Ejemplo}: Con una tasa del $2\%$ mensual, ¿cuánto vale hoy \$ 1.000
    que se recibirán dentro de 1 mes y dentro de 2 meses?

    \begin{equation*}
      \begin{aligned}
        \text{1 mes:} \quad & P = \frac{1.000}{1{,}02} \approx 980{,}39
      \end{aligned}
    \end{equation*}
  \end{frame}

  \begin{frame}
    Cuando hay varios pagos, el valor presente es la suma del valor presente de
    cada uno:
    \[P = C_1(1+i)^{-n_1} + C_2(1+i)^{-n_2} + \cdots + C_k(1+i)^{-n_k}\]

    donde $C_j$ es el pago que ocurre en el periodo $n_j$.
  \end{frame}

  \begin{frame}
    \textit{Definición}: Un flujo de caja es un dibujo, sobre una línea de
    tiempo, de los ingresos (dinero que entra) y los egresos (dinero que sale)
    de una operación financiera.

    \begin{itemize}
      \item El momento $t = 0$ es ``hoy''.
      \item Flecha hacia arriba: dinero que entra.
      \item Flecha hacia abajo: dinero que sale.
      \item Cada pago se ubica en el periodo en que ocurre.
    \end{itemize}
  \end{frame}

  \begin{frame}

    la fecha del pago que no conocemos.

    \begin{itemize}
      \item La deuda original se calcula con su tasa.
      \item Los pagos nuevos se calculan con la tasa del nuevo acuerdo, que puede

      \item Los dos valores se igualan en una misma fecha focal.
    \end{itemize}

  \begin{frame}
    \textit{Planteamiento}: Cuando no conocemos un pago, le ponemos una letra
    (incógnita) y armamos la ecuación de valor. Algunas relaciones frecuentes:

    \begin{itemize}
      \item Un pago es una parte del valor de contado: $C = pV$.
      \item Pagos iguales: $C_1 = C_2 = C$.
      \item Cada pago es una parte del anterior: $C_k = r\,C_{k-1}$.
      \item El saldo final es igual al depósito inicial.
    \end{itemize}
  \end{frame}
